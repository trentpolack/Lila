cmake_minimum_required(VERSION 3.28)
# Also usable without Qt, Python development libraries, or the rest of Lila's dependency graph.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LilaPortabilityTests LANGUAGES CXX)
    enable_testing()
    find_path(LILA_VULKAN_HEADERS vulkan/vulkan_core.h REQUIRED)
    find_path(LILA_DOCTEST_HEADERS doctest/doctest.h REQUIRED)
    find_package(Python REQUIRED COMPONENTS Interpreter)
endif()
get_filename_component(_lila_source "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(lila_portability_tests test_portability.cpp ${_lila_source}/src/core/SystemMemory.cpp)
target_compile_features(lila_portability_tests PRIVATE cxx_std_20)
target_compile_definitions(lila_portability_tests PRIVATE LILA_FOUNDATION_EXPORTS VK_ENABLE_BETA_EXTENSIONS)
target_include_directories(lila_portability_tests PRIVATE ${_lila_source}/src)
if(TARGET Vulkan::Headers)
    target_link_libraries(lila_portability_tests PRIVATE Vulkan::Headers doctest::doctest)
else()
    target_include_directories(lila_portability_tests PRIVATE ${LILA_VULKAN_HEADERS} ${LILA_DOCTEST_HEADERS})
endif()
add_test(NAME lila_portability_tests COMMAND lila_portability_tests)
add_test(NAME lila_python_home COMMAND ${CMAKE_COMMAND} -DLILA_SOURCE=${_lila_source}
    -DPython_EXECUTABLE=${Python_EXECUTABLE} -P ${CMAKE_CURRENT_SOURCE_DIR}/test_python_home.cmake)
add_test(NAME lila_smoke_image_checks COMMAND ${Python_EXECUTABLE} -m unittest discover
    -s ${_lila_source}/python/tests -p test_macos_smoke_image.py)
